import socket
import struct
import time

ETH_P_ALL = 3
# Costanti del kernel Linux per leggere le statistiche dei Raw Socket (da <linux/if_packet.h>)
SOL_PACKET = 263
PACKET_STATISTICS = 6

# --- LIVELLO 2: ETHERNET ---
def format_mac_addr(bytes_addr):
    bytes_str = map('{:02x}'.format, bytes_addr)
    return ':'.join(bytes_str).upper()

def unpack_ethernet_frame(data):
    dest_mac, src_mac, proto = struct.unpack('! 6s 6s H', data[:14])
    return format_mac_addr(dest_mac), format_mac_addr(src_mac), socket.htons(proto), data[14:]

# --- LIVELLO 3: RETE (IPv4) ---
def unpack_ipv4_packet(data):
    version_header_length = data[0]
    version = version_header_length >> 4
    header_length = (version_header_length & 15) * 4
    ttl, proto, src, target = struct.unpack('! 8x B B 2x 4s 4s', data[:20])
    
    def format_ipv4(addr):
        return '.'.join(map(str, addr))
        
    return version, header_length, ttl, proto, format_ipv4(src), format_ipv4(target), data[header_length:]

# --- LIVELLO 4: TRASPORTO (TCP, UDP, ICMP) ---
def unpack_tcp_segment(data):
    (src_port, dest_port, sequence, acknowledgment, offset_reserved_flags) = struct.unpack('! H H L L H', data[:14])
    offset = (offset_reserved_flags >> 12) * 4
    
    flag_urg = (offset_reserved_flags & 32) >> 5
    flag_ack = (offset_reserved_flags & 16) >> 4
    flag_psh = (offset_reserved_flags & 8) >> 3
    flag_rst = (offset_reserved_flags & 4) >> 2
    flag_syn = (offset_reserved_flags & 2) >> 1
    flag_fin = offset_reserved_flags & 1
    
    return src_port, dest_port, sequence, acknowledgment, flag_urg, flag_ack, flag_psh, flag_rst, flag_syn, flag_fin, data[offset:]

def unpack_udp_segment(data):
    src_port, dest_port, size = struct.unpack('! H H 2x H', data[:8])
    return src_port, dest_port, size, data[8:]

def unpack_icmp_packet(data):
    icmp_type, code, checksum = struct.unpack('! B B H', data[:4])
    return icmp_type, code, checksum, data[4:]

# --- TELEMETRIA KERNEL ---
def ottieni_statistiche_kernel(raw_socket):
    """
    Interroga il kernel Linux (struct tpacket_stats) per ottenere:
    - tp_packets: pacchetti totali ricevuti dal socket
    - tp_drops: pacchetti scartati per saturazione del buffer
    """
    try:
        # 8 byte = due unsigned int a 32 bit ('II')
        stats_raw = raw_socket.getsockopt(SOL_PACKET, PACKET_STATISTICS, 8)
        tp_packets, tp_drops = struct.unpack('II', stats_raw)
        return tp_packets, tp_drops
    except OSError as e:
        print(f"Impossibile leggere PACKET_STATISTICS dal kernel: {e}")
        return 0, 0

# --- MOTORE PRINCIPALE ---
def avvia_sniffer(stampa_a_video=True):
    raw_socket = socket.socket(socket.AF_PACKET, socket.SOCK_RAW, socket.ntohs(ETH_P_ALL))
    
    # Dimensione attuale del buffer di ricezione del socket (SO_RCVBUF)
    rcvbuf_size = raw_socket.getsockopt(socket.SOL_SOCKET, socket.SO_RCVBUF)
    print(f"[*] Sniffer avviato | Buffer Socket Kernel (SO_RCVBUF): {rcvbuf_size} byte")
    print(f"[*] Modalità Verbose (print attivo): {stampa_a_video}")
    print("[*] Premi CTRL+C dopo il test di carico per visualizzare il report di Packet Drop...\n")

    pacchetti_processati_python = 0
    start_time = time.time()

    try:
        while True:
            raw_data, address = raw_socket.recvfrom(65535)
            pacchetti_processati_python += 1
            
            dest_mac, src_mac, eth_proto, payload = unpack_ethernet_frame(raw_data)
            
            if eth_proto == 8:
                version, header_length, ttl, proto, src, target, data = unpack_ipv4_packet(payload)
                
                if stampa_a_video:
                    print(f"\n[IPv4] Da: {src} -> A: {target} | Protocollo: {proto}")
                
                if proto == 1:
                    icmp_type, code, checksum, data = unpack_icmp_packet(data)
                    if stampa_a_video:
                        print(f"\t[ICMP] Tipo: {icmp_type} | Codice: {code}")
                        
                elif proto == 6:
                    src_port, dest_port, seq, ack, urg, ack_f, psh, rst, syn, fin, data = unpack_tcp_segment(data)
                    if stampa_a_video:
                        print(f"\t[TCP] Porta Src: {src_port} -> Porta Dest: {dest_port}")
                        print(f"\t[TCP Flags] SYN:{syn} ACK:{ack_f} FIN:{fin} RST:{rst}")
                        
                elif proto == 17:
                    src_port, dest_port, size, data = unpack_udp_segment(data)
                    if stampa_a_video:
                        print(f"\t[UDP] Porta Src: {src_port} -> Porta Dest: {dest_port} | Dimensione: {size}")

    except PermissionError:
        print("Errore critico: Esegui con 'sudo'.")
    except KeyboardInterrupt:
        durata = time.time() - start_time
        tp_packets, tp_drops = ottieni_statistiche_kernel(raw_socket)
        
        # Calcolo percentuale di drop e throughput
        drop_rate = (tp_drops / tp_packets * 100) if tp_packets > 0 else 0.0
        pps = pacchetti_processati_python / durata if durata > 0 else 0.0

        print("\n" + "="*55)
        print(" REPORT PERFORMANCE SNIFFER SINCRONO")
        print("="*55)
        print(f" Durata del test               : {durata:.2f} secondi")
        print(f" Pacchetti processati (Python) : {pacchetti_processati_python}")
        print(f" Velocità di elaborazione      : {pps:.1f} pacchetti/sec (pps)")
        print("-" * 55)
        print(f" Pacchetti visti dal Kernel    : {tp_packets}")
        print(f" Pacchetti persi (Kernel Drop) : {tp_drops}")
        print(f" Tasso di Packet Drop          : {drop_rate:.2f}%")
        print("="*55)
    finally:
        raw_socket.close()

if __name__ == "__main__":
    # Imposta stampa_a_video=True per il primo test, poi prova con False per confrontare!
    avvia_sniffer(stampa_a_video=True)
